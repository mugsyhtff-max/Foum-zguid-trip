      if(entry.isIntersecting){  
        entry.target.classList.add('revealed');  
        revealObserver.unobserve(entry.target);  
      }  
    });  
  }, {threshold:.14});  
  document.querySelectorAll('[data-reveal]').forEach(function(el, idx){  
    el.style.transitionDelay = (idx % 3) * 0.12 + 's';  
    revealObserver.observe(el);  
  });  
  
  /* Animated counters */  
  var runCounter = function(el){  
    var targetVal = parseFloat(el.getAttribute('data-count'));  
    var decimalsVal = parseInt(el.getAttribute('data-decimals') || '0', 10);  
    var startTime = null, durationMs = 2000;  
    var stepFn = function(ts){  
      if(!startTime) startTime = ts;  
      var progress = Math.min((ts - startTime) / durationMs, 1);  
      var eased = 1 - Math.pow(1 - progress, 3);  
      var currentVal = targetVal * eased;  
      if(decimalsVal > 0){  
        el.textContent = (currentVal / Math.pow(10, decimalsVal)).toFixed(decimalsVal);  
      } else {  
        el.textContent = Math.floor(currentVal).toLocaleString('en-US');  
      }  
      if(progress < 1) requestAnimationFrame(stepFn);  
    };  
    requestAnimationFrame(stepFn);  
  };  
  var countObserver = new IntersectionObserver(function(entries){  
    entries.forEach(function(entry){  
      if(entry.isIntersecting){  
        runCounter(entry.target);  
        countObserver.unobserve(entry.target);  
      }  
    });  
  }, {threshold:.6});  
  document.querySelectorAll('.count').forEach(function(el){ countObserver.observe(el); });  
  
  /* Booking form */  
  var bookingFormEl = document.getElementById('bookingForm');  
  var formSuccessEl = document.getElementById('formSuccess');  
  bookingFormEl.addEventListener('submit', function(ev){  
    ev.preventDefault();  
    var nameVal = document.getElementById('bfName').value.trim();  
    var emailVal = document.getElementById('bfEmail').value.trim();  
    var dateVal = document.getElementById('bfDate').value;  
    if(!nameVal || !emailVal || !dateVal){  
      [ 'bfName','bfEmail','bfDate' ].forEach(function(fid){  
        var inputEl = document.getElementById(fid);  
        if(!inputEl.value.trim()){ inputEl.style.borderColor = '#b0563f'; setTimeout(function(){ inputEl.style.borderColor=''; }, 1800); }  
      });  
      return;  
    }  
    bookingFormEl.querySelectorAll('.form-row, .field, button[type="submit"]').forEach(function(el){ el.style.display = 'none'; });  
    formSuccessEl.style.display = 'block';  
  });  
  
  /* Footer year */  
  document.getElementById('yearNow').textContent = new Date().getFullYear();  
})();
